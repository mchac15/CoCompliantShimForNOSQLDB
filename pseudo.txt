Data Structures:
    lock_mode ∈ {SHARED, EXCLUSIVE}

    LockNode: { mode, holders: set of txn, predecessor, next, upgraded }
        // upgraded (default false): the holder read k before writing it, i.e. the node was created or
        // flipped to EXCLUSIVE by upgrade(). A node created by a direct put(k) is a blind write
    chain: { head, tail: LockNode }   // doubly linked through predecessor/next; arrival order = required commit order
        // chain.last_node() = tail. "append n" makes n the new tail (and the head if the chain was empty);
        // "remove n" fixes head/tail if n was one of them. predecessor/next are relinked explicitly by the callers
    locks_map: ConcurrentHashMap<key, chain>   // one chain per key, generations of holders

    txn: { txn_id, locks_acquired: key -> (mode, target_node) (built dynamically), status, code , write_buffer: key -> value }
    status ∈ {started, executed, prepared, committing, committed, aborted, must_abort}

    transactions_map: ConcurrentHashMap<txn_id, txn>
    // atomic(locks_map) { ... } is one critical section over the chains. An implementation may use
    // per-key latches (one at a time) as long as the status re-check in lock()/upgrade() and the update of
    // txn.locks_acquired stay atomic with respect to abort_transaction() reading locks_acquired (e.g. both
    // under a per-txn guard); otherwise a foreign abort_transaction() can miss a node appended concurrently
    lock_timeout: duration   // bounds every wait of the execution phase (lock(), upgrade()); prepare() has none


Store:
    // black box NoSQL store, nothing more is assumed than the following
    store.read(k) -> value
    store.write(k, v)    // blind put; no atomicity across keys assumed

    Assumptions:
        // 1. Blocking semantics: store.write is blocking. If it fails it is simply retried / blocked
        //    until it succeeds, so the protocol has no error path for store failures.
        // 2. No crashes: crash recovery is out of scope. The shim is assumed not to crash between
        //    voting YES and the coordinator's decision (nothing is logged, so a crash there would
        //    lose a prepared txn and break 2PC atomicity).
        // 3. Coordinator: txn_ids are assigned by the coordinator. It calls execute at most once per
        //    txn_id and sends no prepare/commit/abort for a txn_id before its execute has been
        //    delivered. Garbage collection of txn state is its responsibility too: the shim drops a txn
        //    from transactions_map once it is committed or aborted, and later or duplicate messages for
        //    that id take the unknown-id branches (prepare votes NO, commit/abort are no-ops).
        // 4. Deadlocks: only the execution-phase waits (lock(), upgrade()) are bounded, by lock_timeout.
        //    The waits in prepare() have no timeout: a cross-shim cycle at prepare (T1 before T2 on one
        //    shim, T2 before T1 on another) is a 2PC-level deadlock, left to the coordinator (e.g. its
        //    vote timeout) and not handled by the shim.


procedure execute(txn_id, code):
    txn = new txn with txn_id, locks_acquired={}, status=started, code=code
    if transactions_map.putIfAbsent(txn.txn_id, txn) != null
        return FAILED   // txn_id already registered (collision / duplicate)


    execution_failed = false
    txn.code calls get(k, txn) or put(k, v, txn) multiple times

    if during execution, return value of get/put is FAILED, then stop execution
        and set execution_failed = true

    // may block inside a get/put on a predecessor; a FAILED result
    // requires the handler to stop calling get/put and return immediately
    if execution_failed or not CAS(txn.status, started, executed):
        // a get/put that returned FAILED already ran abort_transaction, but the CAS can also fail
        // because a cascade / foreign abort landed after the last get/put, and then nobody has
        // released the locks yet. abort_transaction is idempotent (its CAS fails on an already
        // aborted txn), so always call it here
        abort_transaction(txn)
        return FAILED

    return SUCCEEDED


procedure get(k, txn):
    result = lock(k, SHARED, txn)
    if result == MUST_ABORT:
        abort_transaction(txn)
        return NULL, FAILED
    if k in txn.write_buffer:
        return txn.write_buffer[k], SUCCEEDED   // own write first, no need to look at the chain
    actual_value=null   // null = "no predecessor version", so values are assumed to be non-null
    // speculative read: find the nearest EXCLUSIVE node at or before this txn's
    // own node (never a successor's), i.e. the latest predecessor write that
    // this txn is ordered after. Reading that holder's write_buffer is safe:
    // lock() only returns after every predecessor holder is at least
    // executed, so its buffer is final. If the writer already committed, its
    // buffer was applied to the store before its node was spliced out of the
    // chain, so no such node is found and reading the store is correct.
    atomic(locks_map) {
        if k in locks_map:
            current_node=txn.locks_acquired[k].target_node
            while current_node!=null and current_node.mode!=EXCLUSIVE:
                current_node=current_node.predecessor
            if current_node!=null:
                actual_value=(the sole holder of current_node).write_buffer[k]   // an EXCLUSIVE node has one holder
    }
    if actual_value!=null:
        return actual_value, SUCCEEDED
    return store.read(k), SUCCEEDED


procedure put(k, v, txn):
    result = lock(k, EXCLUSIVE, txn)
    if result == MUST_ABORT:
        abort_transaction(txn)
        return FAILED

    txn.write_buffer[k]=v
    //set value of k to v
    return SUCCEEDED

procedure lock(k, mode, txn):
    if txn.status == aborted || txn.status == must_abort:
        return MUST_ABORT   // txn was cascade-aborted while running

    target_node = null

    if txn.locks_acquired[k] contains (mode, *):
        return ALREADY_HELD
    else if txn.locks_acquired[k] contains (EXCLUSIVE, *):
        return ALREADY_HELD
    else if txn.locks_acquired[k] contains (SHARED, *) and mode == EXCLUSIVE:
        return upgrade(k, txn)

    atomic(locks_map) {
        if txn.status in {aborted, must_abort}:
            // re-check under the latch: a foreign abort_transaction() (coordinator abort(), or
            // prepare() on a txn that is still started) may have released this txn's locks after
            // the check at the top. Appending a node now would leave one nobody ever removes
            return MUST_ABORT

        if k not in locks_map:
            locks_map[k] = new chain

        chain = locks_map[k]
        last_node = chain.last_node()

        if last_node == null:
            target_node = LockNode{mode, holders={txn}, predecessor=null, next=null}
            append target_node to chain
        else if mode == SHARED and last_node.mode == SHARED:
            last_node.holders.add(txn)
            target_node = last_node
        else:
            target_node = LockNode{mode, holders={txn}, predecessor=last_node, next=null}
            last_node.next=target_node
            append target_node to chain

        txn.locks_acquired[k] = (mode, target_node)
    }

    deadline = now() + lock_timeout   // one deadline for the whole acquisition, not reset by the re-read iterations
    loop:
        atomic(locks_map) {
            pred = target_node.predecessor
        }
        if pred == null:
            break

        // pred can be spliced out from under us (by commit()/abort_transaction()
        // on one of its holders) while we wait; re-read target_node.predecessor
        // under the latch on every iteration instead of trusting this captured
        // reference once resolved
        wait until (∀ t in pred.holders,
                t.status in {committed, aborted, executed, prepared, committing, must_abort})
                or txn.status == aborted or txn.status == must_abort
                or pred != target_node.predecessor
                or now() >= deadline

        if pred != target_node.predecessor:
            continue   // pred was spliced out mid-wait; loop back and re-read fresh under the latch

        if ∃ t in pred.holders with t.status == aborted or t.status == must_abort:
            abort_transaction(txn)
            return MUST_ABORT

        if txn.status in {aborted, must_abort}:
            return MUST_ABORT

        if not (∀ t in pred.holders,
                t.status in {committed, aborted, executed, prepared, committing, must_abort}):
            // the wait ended on the deadline: assume a deadlock (or a predecessor that never
            // finishes executing). Always safe: this txn has not executed and voted nothing, so it
            // just aborts. Costs liveness only, never CO or 2PC atomicity
            abort_transaction(txn)
            return MUST_ABORT

        break   // pred fully resolved with no abort among its holders

    // Deadlocks between running txns (T1 holds X and waits for Y while T2 holds Y and waits for X) are
    // broken by lock_timeout above; no wait-for graph is kept

    return LOCK_ACQUIRED


procedure upgrade(k, txn):
    must_wait = false
    node = null
    return_value=MUST_ABORT

    atomic(locks_map) {
        if txn.status in {aborted, must_abort}:
            return MUST_ABORT   // re-check under the latch, same reason as in lock()

        node = txn.locks_acquired[k].target_node
        chain = locks_map[k]

        // Invariant: an upgrade may move this txn's position only across readers, never across a writer
        // (otherwise the value this txn read is stale by the time its own write takes effect).
        // Everything queued behind node is blocked, not running: its holders wait in lock() until every
        // holder of node, this txn included, is executed. So what is behind node is either
        //   - SHARED nodes: only possible after an aborted writer was spliced out of the chain. Their readers
        //     are crossed: the new node goes behind them (skipped by the loop below);
        //   - an EXCLUSIVE node with upgraded == false: a blind write that never read k. Ordering it after our
        //     write is what it wanted anyway, so we go in front of it (it will also wait for us);
        //   - an EXCLUSIVE node with upgraded == true: its holder read k as well. Going in front of it would
        //     make it write a value computed from a stale read (lost update): this txn aborts.
        p = node
        while p.next != null and p.next.mode == SHARED:
            p = p.next
        after = p.next   // EXCLUSIVE or null

        if after != null and after.upgraded:
            // case 3b: read-modify-write against read-modify-write on k: exactly one of the two upgraders
            // has to abort, this one loses. return_value stays MUST_ABORT
            pass

        else if p == node and node.holders == {txn}:
            // case 1: sole holder, at most a blind writer behind: upgrade in place
            node.mode = EXCLUSIVE
            node.upgraded = true
            txn.locks_acquired[k] = (EXCLUSIVE, node)
            return_value = UPGRADED

        else:
            // cases 2 and 3a: leave node and take a new EXCLUSIVE node right behind p: at the tail if nothing
            // is behind p (case 2), otherwise in front of the blind writer `after` (case 3a)
            node.holders.remove(txn)
            new_node = LockNode{EXCLUSIVE, holders={txn}, predecessor=p, next=after, upgraded=true}
            p.next = new_node
            if after != null:
                after.predecessor = new_node   // its wait loops re-read predecessor, so it now also waits for us
            else:
                append new_node to chain
            if node.holders is empty:   // only possible when p != node
                if node.predecessor != null:
                    node.predecessor.next=node.next
                if node.next != null:
                    node.next.predecessor=node.predecessor
                remove node from chain
            txn.locks_acquired[k] = (EXCLUSIVE, new_node)
            must_wait = true
            return_value = UPGRADED
    }

    if must_wait:
        deadline = now() + lock_timeout
        loop:
            atomic(locks_map) {
                pred = new_node.predecessor
            }
            if pred == null:
                break

            // pred (the readers we are queued behind) can be spliced out from
            // under us while we wait; re-read
            // new_node.predecessor under the latch on every iteration instead
            // of trusting a captured reference once resolved
            wait until (∀ t in pred.holders,
                        t.status in {committed, aborted, executed, prepared, committing, must_abort})
                        or txn.status == aborted or txn.status == must_abort
                        or pred != new_node.predecessor
                        or now() >= deadline

            if pred != new_node.predecessor:
                continue   // pred was spliced out mid-wait; loop back and re-read fresh under the latch

            if ∃ t in pred.holders with t.status == aborted or t.status == must_abort:
                abort_transaction(txn)
                return MUST_ABORT

            if txn.status in {aborted, must_abort}:
                return MUST_ABORT

            if not (∀ t in pred.holders,
                        t.status in {committed, aborted, executed, prepared, committing, must_abort}):
                abort_transaction(txn)   // deadline reached: same reasoning as in lock()
                return MUST_ABORT

            break   // pred fully resolved with no abort among its holders

        // the wait above is bounded by lock_timeout, like in lock(): e.g. two readers of the same
        // shared node, one upgrading and waiting for the other, which waits on this txn via another key

    return return_value


procedure prepare(txn_id):
    txn = transactions_map[txn_id]
    if txn == null:
        // unknown/already-evicted txn_id: treat as a duplicate/late
        // prepare for a txn that was already resolved and cleaned up
        send no to 2PC controller
        return

    if txn.status in {prepared, committing, committed}:
        // duplicate/late prepare (coordinator messages are idempotent): the txn already voted YES
        // and every predecessor was resolved before that, so just re-send the vote
        send yes to 2PC controller
        return

    if txn.status not in {executed, prepared}:
        // covers the already-aborted case too: cascade abort sets
        // txn.status directly, so no predecessor dereference is needed here
        send no to 2PC controller
        abort_transaction(txn)
        return

    for each (k, mode, node) in txn.locks_acquired:
        loop:
            atomic(locks_map) {
                pred = node.predecessor
            }
            if pred == null:
                break

            // pred can be spliced out from under us while we wait; re-read
            // node.predecessor under the latch on every iteration instead of
            // trusting this captured reference once resolved
            wait until (∀ t in pred.holders, t.status in {committed, aborted, must_abort})
                        or txn.status == aborted or txn.status == must_abort
                        or pred != node.predecessor
            // txn may have been cascade-aborted by a different key's
            // predecessor while waiting here

            if txn.status in {aborted, must_abort}:
                send no to 2PC controller

                if txn.status == must_abort:
                    abort_transaction(txn)
                return

            if pred != node.predecessor:
                continue   // pred was spliced out mid-wait; loop back and re-read fresh under the latch

            if ∃ t in pred.holders with t.status == aborted or t.status == must_abort:
                abort_transaction(txn)
                send no to 2PC controller
                return

            break   // pred fully resolved with no abort among its holders

    if not CAS(txn.status, {executed, prepared}, prepared):
        if txn.status in {prepared, committing, committed}:
            // a concurrent duplicate prepare already CASed to prepared and voted YES (and the
            // coordinator may even have moved on to commit). Voting NO now would contradict that
            // vote, so just re-send it
            send yes to 2PC controller
            return
        // status moved out from under us (e.g. a late cascade mark_must_abort
        // landed after the wait loop finished) --- the txn never became
        // prepared, so it is always safe to abort it here and release its
        // locks rather than leaking them
        send no to 2PC controller
        abort_transaction(txn)
        return

    send yes to 2PC controller


procedure commit(txn_id):
    txn = transactions_map[txn_id]
    if txn == null:
        return   // unknown/already-evicted txn_id: no-op

    // committing (not committed) while the store is being written: a coordinator abort can no
    // longer take the txn (abort_transaction's CAS excludes it), and a successor's prepare cannot
    // pass before the writes are applied, since this txn is only spliced out of the chains below
    if not CAS(txn.status, prepared, committing):
        return

    for (key,value) in txn.write_buffer:
        store.write(key, value)

    atomic(locks_map) {
        for each (k, mode, target_node) in txn.locks_acquired:
            chain = locks_map[k]
            remove txn from target_node.holders
            if target_node.holders is empty:
                if target_node.predecessor != null:
                    target_node.predecessor.next=target_node.next
                if target_node.next != null:
                    target_node.next.predecessor=target_node.predecessor
                remove target_node from chain
                if chain is empty:
                    remove k from locks_map   // under the latch: a concurrent lock(k) sees the chain or none
    }
    txn.status=committed
    transactions_map.remove(txn.txn_id)
    // resolved: the volatile state is dropped, nothing in locks_map references the txn anymore.
    // Whatever comes for this txn_id afterwards is handled by the unknown-id branches
    // (assumption 3: the coordinator owns the txn_id lifecycle)


procedure abort_transaction(txn):
    if not CAS(txn.status, {executed, prepared, must_abort, started}, aborted):
        return

    doomed_finished = {}   // txn_ids of cascade victims that had already finished executing
    atomic(locks_map) {
        for each (k, mode, target_node) in txn.locks_acquired:
            chain = locks_map[k]

            if mode == EXCLUSIVE:
                successor_node = target_node.next
                while successor_node != null:
                    for each t in successor_node.holders:
                        if mark_must_abort(t.txn_id):
                            doomed_finished.add(t.txn_id)
                    successor_node = successor_node.next

            remove txn from target_node.holders
            if target_node.holders is empty:
                if target_node.predecessor != null:
                    target_node.predecessor.next=target_node.next
                if target_node.next != null:
                    target_node.next.predecessor=target_node.predecessor
                remove target_node from chain
                if chain is empty:
                    remove k from locks_map   // same conditional removal as in commit()
    }


    transactions_map.remove(txn.txn_id)
    // resolved: evicted like in commit(). Successors observe txn.status directly (they hold a
    // reference, not a map lookup), and a duplicate/late message for the evicted id is a no-op abort

    // eager cleanup, outside the latch (so at most one key latch is ever held): a doomed txn that
    // already finished executing has no thread left that would ever abort it, and waiting for the
    // coordinator would leave its nodes in the chain (every new writer behind it aborts, and its own
    // successors are not doomed until it aborts). We go through abort(id) here, the same txn_id-based
    // entry point a foreign caller uses, instead of holding onto the txn object and calling
    // abort_transaction(t) directly: by the time this runs the victim may already have been evicted
    // by another cascade or by the coordinator, and abort(id)'s map lookup makes that the ordinary
    // unknown/already-evicted no-op instead of a stale-reference special case. Aborting it now
    // releases its locks and cascades transitively; its later prepare/commit/abort hit the
    // unknown-id branches (prepare votes NO)
    for each id in doomed_finished:
        abort(id)

procedure abort(txn_id):
    txn = transactions_map[txn_id]
    if txn == null:
        return   // unknown/already-evicted txn_id: no-op, per the
                 // evicted-id convention documented in abort_transaction()
    abort_transaction(txn)


procedure mark_must_abort(txn_id) -> bool:
    // returns true iff txn was executed: nobody runs it anymore, so the caller has to clean it up
    // itself (see abort_transaction). Never touches prepared
    txn = transactions_map[txn_id]

    if txn == null:
        return false   // unknown/already-evicted txn_id: no-op, per the
                        // evicted-id convention documented in abort_transaction()
    if CAS(txn.status, executed, must_abort):
        return true
    CAS(txn.status, started, must_abort)   // still running: its own thread aborts it at its next
                                            // lock() or at the CAS in execute()
    return false
