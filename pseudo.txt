Data Structures:
    lock_mode ∈ {SHARED, EXCLUSIVE}

    LockNode: { mode, holders: set of txn, granted: set of txn, predecessor, next, upgraded }
        // one generation of holders of a key: consecutive SHARED requests share a node, every EXCLUSIVE
        // request (or a SHARED one behind an EXCLUSIVE node) gets a new one, so an EXCLUSIVE node has one holder
        // granted ⊆ holders: the holders that got past the wait in lock(), i.e. that may have read the key.
        //   A holder still blocked in lock() read nothing yet, so a cascade spares it (see abort_transaction)
        // upgraded: the holder read k before writing it (upgrade() or get_for_update). A node from a plain
        //   put(k) is a blind write: it depends on no earlier value of k
    chain: { head, tail: LockNode }   // doubly linked through predecessor/next; order = required commit order
    locks_map: ConcurrentHashMap<key, chain>   // one chain per key; a key k stands for (table, key)

    txn: { txn_id, ts, status, locks_acquired: key -> (mode, node), write_buffer: key -> value,
           read_buffer: key -> value, reached_executed, depends_on: set of txn, dependents, fanout_cap }
        // ts: the txn's age (smaller = older), taken from the global transaction id, so every shim agrees on
        //     it. A retry is a new global transaction, hence younger
        // read_buffer: the value each get(k) returned, so a repeated get(k) returns the same value
        // reached_executed: the txn was executed at some point, so a successor may have read its write_buffer
        // depends_on / dependents / fanout_cap: the fanout cap's bookkeeping (see admit_fanout)
    status ∈ {started, executed, prepared, committing, committed, aborted, must_abort}
    SPEC_READY = {executed, prepared, committing, committed}
        // the speculation point: from executed on, a txn's write_buffer no longer changes, so successors may
        // read on top of it. The non-speculative baseline is the same protocol with SPEC_READY = {committed}
        // (no successor ever reads an uncommitted value, no abort ever cascades, no fanout cap)

    transactions_map: ConcurrentHashMap<txn_id, txn>
        // also holds the tombstones: an aborted txn with no locks and an expires_at, left by an abort that
        // came before the start (see abort()), removed after tombstone_ttl
    F: real >= 0   // the shim's adaptive fanout cap, tuned by fanout_controller()
    lock_timeout   // backstop on the waits of lock()/upgrade(); deadlocks are prevented by wound-wait
    // atomic(locks_map) { ... } is one critical section over the chains (an implementation may use a
    // latch per key, one at a time). "wait until" sleeps until something changes on the chain it reads


Store:
    store.read(k) -> value
    store.write(k, v)   // blind put, blocking, no atomicity across keys

    Assumptions:
        // 1. No crashes: crash recovery is out of scope (nothing is logged).
        // 2. The coordinator assigns txn_ids, calls start once per txn_id, and sends get/put/end of one txn
        //    one at a time (the client driver serializes them). abort/prepare/commit may arrive at any time,
        //    an abort even before the start (the shim then leaves a tombstone).
        // 3. Deadlocks: execution-phase waits are made acyclic by wound-wait, prepare-phase waits by
        //    wait-die (both on the global ages ts). lock_timeout only remains as a backstop.


procedure start(txn_id):
    txn = new txn with txn_id, ts = age(txn_id), status = started, everything else empty
    if transactions_map.putIfAbsent(txn_id, txn) != null:
        return FAILED   // duplicate, or tombstoned: the coordinator already aborted this id
    return SUCCEEDED


procedure end(txn_id):
    // the client sent its last get/put: the write_buffer is final from here on
    txn = transactions_map[txn_id]
    if txn == null:
        return FAILED
    if not CAS(txn.status, started, executed):
        if txn.status in {executed, prepared, committing, committed}:
            return SUCCEEDED   // duplicate end
        abort_transaction(txn)   // doomed while running: release its locks (idempotent)
        return FAILED
    txn.reached_executed = true
    return SUCCEEDED


procedure get(txn_id, k, for_update = false):
    // for_update = get_for_update(k), i.e. SELECT ... FOR UPDATE: the txn will write k after reading it, so
    // it takes the EXCLUSIVE lock right away. The later put(k) then needs no upgrade, and two read-modify-
    // writes of k queue one behind the other instead of one of them aborting in upgrade()
    txn = transactions_map[txn_id]
    if txn == null or txn.status in {executed, prepared, committing, committed}:
        return NULL, FAILED   // unknown txn_id, or a request after end()
    if for_update:
        result = lock(k, EXCLUSIVE, txn, reads_key = true)
    else:
        result = lock(k, SHARED, txn)
    if result == MUST_ABORT:
        abort_transaction(txn)
        return NULL, FAILED
    if k in txn.write_buffer:
        return txn.write_buffer[k], SUCCEEDED   // own write first

    value = null   // null = no predecessor version (values are non-null)
    atomic(locks_map) {
        if txn.status in {aborted, must_abort}:
            doomed = true   // a foreign abort released this txn's locks since lock() returned
        else if k in txn.read_buffer:
            value = txn.read_buffer[k]   // the version visible to txn cannot change while it holds k
        else:
            // speculative read: the write of the nearest EXCLUSIVE node ahead of txn's own node whose
            // holder wrote k (a get_for_update holder that did not write k holds no version of it).
            // Its write_buffer is final: lock() only returned once every holder ahead was executed
            for n = txn.locks_acquired[k].node.predecessor; n != null; n = n.predecessor:
                if n.mode == EXCLUSIVE and k in (the holder of n).write_buffer:
                    value = (the holder of n).write_buffer[k]
                    break
    }
    if doomed:
        abort_transaction(txn)
        return NULL, FAILED
    if value == null:
        value = store.read(k)   // no uncommitted version ahead: the store holds the latest committed one
    txn.read_buffer[k] = value
    return value, SUCCEEDED


procedure put(txn_id, k, v):
    txn = transactions_map[txn_id]
    if txn == null or txn.status in {executed, prepared, committing, committed}:
        return FAILED
    if lock(k, EXCLUSIVE, txn) == MUST_ABORT:
        abort_transaction(txn)
        return FAILED
    txn.write_buffer[k] = v
    return SUCCEEDED


procedure lock(k, mode, txn, reads_key = false):
    held = txn.locks_acquired[k]
    if held != null:
        if held.mode == EXCLUSIVE or mode == SHARED:
            return ALREADY_HELD if txn.status == started else MUST_ABORT
        return upgrade(k, txn)   // holds k SHARED, wants it EXCLUSIVE

    atomic(locks_map) {
        // checked under the latch: a foreign abort_transaction may have released txn's locks, and a node
        // appended now would never be removed
        if txn.status != started:
            return MUST_ABORT
        chain = locks_map[k], created if absent
        if mode == SHARED and chain.tail != null and chain.tail.mode == SHARED:
            node = chain.tail
            node.holders.add(txn)
        else:
            node = LockNode{mode, holders = {txn}, granted = {}, upgraded = reads_key}
            append node to chain
        txn.locks_acquired[k] = (mode, node)
    }
    return await_grant(k, txn, node)


procedure upgrade(k, txn):
    // SHARED -> EXCLUSIVE. Invariant: txn only moves across readers, never across a writer (its read would
    // be stale by the time its own write takes effect: lost update)
    atomic(locks_map) {
        if txn.status != started:
            return MUST_ABORT
        node = txn.locks_acquired[k].node
        p = node
        while p.next != null and p.next.mode == SHARED:   // skip the readers queued behind node
            p = p.next
        after = p.next   // EXCLUSIVE or null
        if after != null and after.upgraded:
            return MUST_ABORT   // read-modify-write against read-modify-write: this one loses
        if p == node and node.holders == {txn}:
            node.mode = EXCLUSIVE   // sole holder, at most a blind writer behind: flip in place
            node.upgraded = true
            txn.locks_acquired[k] = (EXCLUSIVE, node)
            return LOCK_ACQUIRED
        // leave node, take a new EXCLUSIVE node right behind p: at the tail, or in front of the blind
        // writer `after` (which then also waits for txn)
        new_node = LockNode{EXCLUSIVE, holders = {txn}, upgraded = true,
                            granted = {txn} if txn in node.granted else {}}   // it read k: a cascade must reach it
        remove txn from node.holders (and node.granted); splice node out if it is now empty
        insert new_node between p and after
        txn.locks_acquired[k] = (EXCLUSIVE, new_node)
    }
    return await_grant(k, txn, new_node)   // wait for the readers it was queued behind


procedure await_grant(k, txn, node):
    // the wait of lock()/upgrade(): until every holder ahead reached the speculation point and the grant
    // fits the fanout cap. Wound-wait keeps it deadlock-free: an older txn never waits for a younger running
    // one, it aborts it instead, so every wait-for edge between running txns goes from younger to older
    deadline = now() + lock_timeout
    loop:
        victims = {}
        atomic(locks_map) {
            if txn.status in {aborted, must_abort}:
                return MUST_ABORT
            pred = node.predecessor   // re-read every time: nodes ahead are spliced out / inserted meanwhile
            if pred == null or ∀ t in pred.holders: t.status in SPEC_READY:
                if admit_fanout(txn, node):
                    node.granted.add(txn)
                    return LOCK_ACQUIRED
                // refused: wait, as without speculation, for the full writer to commit or lose dependents
            else:
                // wound-wait: the holders ahead that block txn (not yet executed, not already leaving)
                blocking = {t ∈ pred.holders : t.status not in SPEC_READY ∪ {must_abort, aborted}}
                if blocking != {} and ∀ t in blocking: t.status == started and t.ts > txn.ts:
                    for t in blocking:
                        // a victim is still started: nobody read its buffer, it voted nothing
                        if CAS(t.status, started, must_abort):
                            victims.add(t)
                // an older holder (or a younger one already executed/prepared) is simply waited for
        }
        if victims != {}:
            for t in victims:
                abort_transaction(t)   // outside the latch; its release splices its node out
            continue
        wait until something changes on chain k or now() >= deadline
        if now() >= deadline:
            return MUST_ABORT   // backstop only (e.g. a stalled holder)


procedure risky_ahead(node) -> list of txn:
    // the txns whose abort would cascade to node's holder: the undecided holders of the EXCLUSIVE nodes
    // ahead of it, back to the first blind write (a blind write depends on nobody's value)
    if node.mode == EXCLUSIVE and not node.upgraded:
        return {}
    risky = {}
    for n = node.predecessor; n != null; n = n.predecessor:
        if n.mode == EXCLUSIVE:
            w = the holder of n
            if w.status not in {committing, committed}:
                risky.add(w)
            if not n.upgraded:
                break
    return risky


procedure admit_fanout(txn, node) -> bool:
    // fanout cap: each writer has at most fanout_cap running dependents, i.e. txns whose grant reads on top
    // of it (a dependent counts once, through however many keys). Bounds how many txns one abort can doom.
    // Under the latch, all or nothing: returns true if txn may be granted
    risky = risky_ahead(node) \ txn.depends_on
    for w in risky:
        if w.fanout_cap is unset:
            w.fanout_cap = floor(F) + (1 with probability F - floor(F))   // drawn once: F on average
    full = {w ∈ risky : w.dependents >= w.fanout_cap}
    if full != {} and ∀ w in full: w.ts < txn.ts:
        return false   // refused: txn waits for an older writer, like without speculation
    // admitted, or forced: a refused grant would wait for a younger txn's commit, which may itself wait
    // (in prepare() on another shim) for this txn: a cycle. Then it is granted past the cap
    for w in risky:
        w.dependents++
        txn.depends_on.add(w)
    return true


procedure release_dependencies(txn):
    // txn is resolved: it no longer counts as a dependent. Wakes the grants refused on those writers
    for w in txn.depends_on:
        w.dependents--
    txn.depends_on = {}


procedure prepare(txn_id):
    txn = transactions_map[txn_id]
    if txn == null:
        send no to 2PC controller
        return
    if txn.status in {prepared, committing, committed}:
        send yes to 2PC controller   // duplicate prepare: re-send the vote
        return
    if txn.status != executed:
        send no to 2PC controller   // still running, aborted or doomed
        abort_transaction(txn)
        return

    // CO: vote only once every txn ahead of txn on every one of its keys has committed
    for each (k, mode, node) in txn.locks_acquired:
        loop:
            atomic(locks_map) {
                pred = node.predecessor   // re-read every time: nodes ahead are spliced out meanwhile
                ready = pred == null or ∀ t in pred.holders: t.status == committed
                // wait-die: with speculation two shims can order the same two txns oppositely, and each
                // prepare then waits for the other's commit, i.e. for the other's vote. Waiting only for
                // older txns (global ages: the same order on every shim) makes such a cycle impossible.
                // Holders that are committing or already leaving resolve without waiting: waited for
                die = txn.status in {aborted, must_abort}   // cascade-doomed while waiting
                      or (not ready and ∃ t in pred.holders:
                              t.ts > txn.ts and t.status not in {committing, committed, must_abort, aborted})
            }
            if die:
                send no to 2PC controller
                abort_transaction(txn)
                return
            if ready:
                break
            wait until something changes on chain k

    if not CAS(txn.status, executed, prepared):
        if txn.status in {prepared, committing, committed}:
            send yes to 2PC controller   // a concurrent duplicate prepare already voted YES
            return
        send no to 2PC controller
        abort_transaction(txn)
        return
    send yes to 2PC controller   // prepared is final: no cascade or wound can take txn anymore


procedure commit(txn_id):
    txn = transactions_map[txn_id]
    if txn == null or not CAS(txn.status, prepared, committing):
        return   // unknown, duplicate, or a tombstone
    // committing: decided, so txn no longer counts in risky_ahead (wakes the grants it held back)
    for (k, v) in txn.write_buffer:
        store.write(k, v)
    atomic(locks_map) {
        for each (k, mode, node) in txn.locks_acquired:
            remove txn from node; splice node out if it is now empty; remove k from locks_map if empty
    }
    txn.status = committed
    release_dependencies(txn)
    commits++   // for fanout_controller()
    transactions_map.remove(txn_id)


procedure abort_transaction(txn):
    if not CAS(txn.status, {started, executed, must_abort, prepared}, aborted):
        return   // already aborted, or committing/committed
    doomed_finished = {}
    atomic(locks_map) {
        for each (k, mode, node) in txn.locks_acquired:
            // cascade before splice: the dependents are marked before txn leaves the chain, so none of
            // them can get past txn's node and vote YES on its rolled-back write.
            // Only a txn that reached executed can have been read by a successor (lock() waits for that);
            // so a wounded txn, or one the coordinator aborted while it ran, dooms nobody
            if mode == EXCLUSIVE and txn.reached_executed:
                for s = node.next; s != null; s = s.next:
                    if s.mode == EXCLUSIVE and not s.upgraded:
                        break   // blind write: does not depend on txn's value, nor do the nodes behind it
                    for t in s.granted:   // holders still blocked in lock() read nothing: spared
                        if mark_must_abort(t):
                            doomed_finished.add(t)
            remove txn from node; splice node out if it is now empty; remove k from locks_map if empty
    }
    release_dependencies(txn)
    transactions_map.remove(txn.txn_id)
    // a doomed txn that already executed has no thread left to abort it: do it now (cascades transitively)
    for t in doomed_finished:
        abort_transaction(t)


procedure mark_must_abort(txn) -> bool:
    // never touches prepared (a YES vote is final). A started victim aborts itself at its next request
    before = CAS_returning(txn.status, {started, executed}, must_abort)
    return before == executed


procedure abort(txn_id):
    // known txn: abort it. Unknown id: leave a tombstone (status aborted, no locks), atomically with
    // start()'s putIfAbsent, so that a start arriving after the abort fails instead of running
    txn = transactions_map.computeIfAbsent(txn_id, new txn with status = aborted, expires_at = now() + tombstone_ttl)
    abort_transaction(txn)   // a no-op on a tombstone


procedure gc_tombstones():   // background
    remove from transactions_map every tombstone past its expires_at


procedure fanout_controller():   // background, one per shim
    // hill-climbs the fractional fanout cap F on the shim's commit rate. One step changes the rate by
    // less than the noise, so each period is compared with an anchor (the last F that made a clear
    // difference) rather than with the previous period: small effects add up until they show
    direction = random ±1
    anchor_F, anchor_rate = F, commit rate of the first period
    every period:
        rate = commits per second over the period, without its first moments (still under the old F)
        if rate > anchor_rate * (1 + eps):
            anchor_F, anchor_rate = F, rate
            F = step(F, direction, big)     // clearly better: go on, faster
        else if rate < anchor_rate * (1 - eps):
            F = anchor_F                    // clearly worse: back to the anchor, turn around
            direction = -direction
        else if too many small steps since the anchor:
            anchor_F, anchor_rate = F, rate // F barely matters here: re-anchor, turn around
            direction = -direction
        else:
            F = step(F, direction, small)   // no clear effect yet: go on, slowly
    // step(F, d, s) = F * (1 + s) if d > 0 else F / (1 + s), at most F_max; below F_min it becomes 0
    // (no speculation but the forced grants), and a step up from 0 goes back to F_min
