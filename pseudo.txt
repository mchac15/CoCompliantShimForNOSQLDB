Data Structures:
    lock_mode ∈ {SHARED, EXCLUSIVE}

    LockNode: { mode, holders: set of txn, predecessor, next }
    // TODO: integrate this data structure
    locks_map: ConcurrentHashMap<key, LinkedList<LockNode>>

    locks_map: key -> chain (linked list of LockNode)   // one chain per key, generations of holders

    txn: { txn_id, locks_acquired: key -> (mode, target_node) (built dynamically), status, code }
    status ∈ {executed, prepared, committed, aborted, must_abort, abort_requested, NULL}
    // abort_requested: "running and doomed" --- a foreign thread (the 2PC
    // coordinator via abort()/prepare(), or a cascade via mark_must_abort)
    // wants this txn aborted while it may still be NULL (its own executor
    // thread still running client code). Only that executor thread may act
    // on it, per the single-cleanup-owner invariant; see lock()/execute().
    // TODO: intermediate states for retry mechanism

    transactions_map: ConcurrentHashMap<txn_id, txn>

    // tombstones for abort() calls that arrive before execute() has
    // registered their txn_id in transactions_map. Not garbage collected
    // for txn_ids that never call execute(); a known, deferred leak.
    aborted_before_start: ConcurrentHashMap<txn_id, boolean>


procedure execute(code):
    txn = new txn with txn_id, locks_acquired={}, status=NULL, code=code
    if transactions_map.putIfAbsent(txn.txn_id, txn) != null
        return FAILED   // txn_id already registered (collision / duplicate)

    if aborted_before_start.remove(txn.txn_id) != null:
        // abort(txn_id) arrived before we registered above; locks_acquired
        // is still empty so there is nothing to splice out of any chain,
        // abort_transaction(txn) just performs the status transition
        abort_transaction(txn)
        return FAILED

    execution_failed = false
    txn.code calls get(k, txn) or put(k, v, txn) multiple times

    if during execution, return value of get/put is FAILED, then stop execution
        and set execution_failed = true

    // may block inside a get/put on a predecessor; a FAILED result
    // requires the handler to stop calling get/put and return immediately
    if execution_failed or not CAS(txn.status, NULL, executed):
        // status may have left NULL some other way while we ran: a foreign
        // abort() / prepare() / cascade set abort_requested. We are still
        // the sole owner of this txn's locks/writes (single-cleanup-owner
        // invariant), so we perform the actual cleanup ourselves, here.
        if txn.status == abort_requested:
            abort_transaction(txn)
        return FAILED

    return SUCCEEDED


procedure get(k, txn):
    result = lock(k, SHARED, txn)
    if result == MUST_ABORT:
        abort_transaction(txn)
        return NULL, FAILED
    return value of k, SUCCEEDED


procedure put(k, v, txn):
    result = lock(k, EXCLUSIVE, txn)
    if result == MUST_ABORT:
        abort_transaction(txn)
        return FAILED
    set value of k to v
    return SUCCEEDED

procedure lock(k, mode, txn):
    if txn.status == aborted || txn.status == must_abort || txn.status == abort_requested:
        return MUST_ABORT   // txn was cascade-aborted, or foreign-abort-requested, while running

    target_node = null

    if txn.locks_acquired[k] contains (mode, *):
        return ALREADY_HELD
    else if txn.locks_acquired[k] contains (EXCLUSIVE, *):
        return ALREADY_HELD
    else if txn.locks_acquired[k] contains (SHARED, *) and mode == EXCLUSIVE:
        return upgrade(k, txn)

    atomic(locks_map) {
        if k not in locks_map:
            locks_map[k] = new chain

        chain = locks_map[k]
        last_node = chain.last_node()

        if txn.status in {aborted, must_abort, abort_requested}:
            // re-check under the key latch: a foreign abort request may
            // have landed between the top-of-function check and here; do
            // not let an already-doomed txn join the chain
            target_node = null
        else if last_node == null:
            target_node = LockNode{mode, holders={txn}, predecessor=null, next=null}
            append target_node to chain
        else if mode == SHARED and last_node.mode == SHARED:
            last_node.holders.add(txn)
            target_node = last_node
        else:
            target_node = LockNode{mode, holders={txn}, predecessor=last_node, next=null}
            last_node.next=target_node
            append target_node to chain

        if target_node != null:
            txn.locks_acquired[k] = (mode, target_node)
    }

    if target_node == null:
        return MUST_ABORT

    if target_node.predecessor != null:
        wait until (∀ t in target_node.predecessor.holders,
                t.status in {committed, aborted, executed, prepared, must_abort})
                or txn.status in {aborted, must_abort, abort_requested}

        if ∃ t in target_node.predecessor.holders with t.status == aborted or t.status == must_abort:
            abort_transaction(txn)
            return MUST_ABORT

        if txn.status in {aborted, must_abort, abort_requested}:
            return MUST_ABORT

        // TODO: no deadlock detection yet --- this wait is unbounded and
        // can deadlock against another txn waiting on this one; accepted
        // for now, to be addressed later

    return LOCK_ACQUIRED


procedure upgrade(k, txn):
    if txn.status == aborted || txn.status == must_abort || txn.status == abort_requested:
        return MUST_ABORT   // txn was cascade-aborted, or foreign-abort-requested, while running

    must_wait = false
    node = null
    return_value=MUST_ABORT

    atomic(locks_map) {
        node = txn.locks_acquired[k].target_node

        if node == locks_map[k].last_node() and node.holders == {txn}:
            locks_map[k].last_node().mode = EXCLUSIVE
            txn.locks_acquired[k] = (EXCLUSIVE, node)
            return_value = UPGRADED

        else if node == locks_map[k].last_node() and node.holders ⊇ {txn}:
            node.holders.remove(txn)
            new_node = LockNode{EXCLUSIVE, holders={txn}, predecessor=node, next=null}
            locks_map[k].last_node().next=new_node
            new_node.predecessor=node
            node.next = new_node
            txn.locks_acquired[k] = (EXCLUSIVE, new_node)
            must_wait = true
            return_value = UPGRADED

        else:
            // case 3: something newer is already queued behind this txn's
            // shared node (it's no longer the tail). Deliberately aborts
            // ordinary read-modify-write patterns for now; a known, accepted
            // limitation (no lock-conversion/queue-jumping support yet),
            // not a bug in this control flow. return_value stays MUST_ABORT.
    }

    if must_wait:
        wait until (∀ t in node.holders,
                    t.status in {committed, aborted, executed, prepared, must_abort})
                    or txn.status in {aborted, must_abort, abort_requested}

        if ∃ t in target_node.predecessor.holders with t.status == aborted or t.status == must_abort:
            abort_transaction(txn)
            return MUST_ABORT

        if txn.status in {aborted, must_abort, abort_requested}:
            return MUST_ABORT
        // TODO: same unbounded-wait / no-deadlock-detection caveat as in lock()

    return return_value 


procedure prepare(txn_id):
    txn = transactions_map[txn_id]
    if txn == null:
        // unknown/already-evicted txn_id: treat as a duplicate/late
        // prepare for a txn that was already resolved and cleaned up
        send no to 2PC controller
        return

    if txn.status == committed:
        // duplicate/late prepare after commit already happened: nothing
        // to abort, txn is already resolved the other way
        send no to 2PC controller
        return

    if txn.status == NULL and CAS(txn.status, NULL, abort_requested):
        // txn is still executing on its own thread. Only that thread may
        // touch its locks_acquired/store writes (single-cleanup-owner
        // invariant), so we do NOT call abort_transaction here --- the
        // executor will notice abort_requested (in lock()/execute()) and
        // clean itself up.
        send no to 2PC controller
        return

    if txn.status not in {executed, prepared}:
        // covers the already-aborted/must_abort case, and a concurrently
        // set abort_requested we lost the race to set ourselves (e.g. a
        // cascade via mark_must_abort beat us to it)
        send no to 2PC controller
        if txn.status != abort_requested:
            abort_transaction(txn)
        return

    for each (k, mode, node) in txn.locks_acquired:
        if node.predecessor != null:
            wait until (∀ t in node.predecessor.holders, t.status in {committed, aborted, must_abort})
                        or txn.status == aborted or txn.status == must_abort
            // txn may have been cascade-aborted by a different key's
            // predecessor while waiting here

            if txn.status in {aborted, must_abort}:
                send no to 2PC controller

                if txn.status == must_abort:
                    abort_transaction(txn)
                return

            if ∃ t in node.predecessor.holders with t.status == aborted or t.status == must_abort:
                abort_transaction(txn)
                send no to 2PC controller
                return

    if not CAS(txn.status, {executed, prepared}, prepared):
        // status moved out from under us (e.g. a late cascade mark_must_abort
        // landed after the wait loop finished) --- the txn never became
        // prepared, so it is always safe to abort it here and release its
        // locks rather than leaking them
        send no to 2PC controller
        abort_transaction(txn)
        return

    send yes to 2PC controller


procedure commit(txn_id):
    txn = transactions_map[txn_id]
    if txn == null:
        return   // unknown/already-evicted txn_id: no-op

    if not CAS(txn.status, prepared, committed):
        return

    atomic(locks_map) {
        for each (k, mode, target_node) in txn.locks_acquired:

            remove txn from target_node.holders
            if target_node.holders is empty:
                if target_node.predecessor != null:
                    tartget_node.predecessor.next=target_node.next
                if target_node.next != null:
                    target_node.next.predecessor=target_node.predecessor
                remove target_node from chain
    }
    // safe to evict txn from any external txn-id registry after this
    // point --- nothing in locks_map references it anymore


procedure abort_transaction(txn):
    if not CAS(txn.status, {executed, prepared, NULL, must_abort, abort_requested}, aborted):
        return
    // NULL and abort_requested sources are only ever reached here via the
    // txn's own executor thread (execute()'s tombstone/post-code paths,
    // lock()'s self-checks) --- a foreign thread never calls this directly
    // while status could still be NULL, see abort()/prepare()


    atomic(locks_map) {
        for each (k, mode, target_node) in txn.locks_acquired:
            entry = locks_map[k]
            if entry == null:
                continue

            if mode == EXCLUSIVE:
                successor_node = target_node.next
                while successor_node != null:
                    for each t in successor_node.holders:
                        mark_must_abort(t)
                    successor_node = successor_node.next

            remove txn from target_node.holders
            if target_node.holders is empty:
                if target_node.predecessor != null:
                    tartget_node.predecessor.next=target_node.next
                if target_node.next != null:
                    target_node.next.predecessor=target_node.predecessor
                remove target_node from locks_map[k] 
    }


    // safe to evict txn from any external txn-id registry after this
    // point --- status is already set for any successor to observe
    // directly, and a duplicate/late message for an evicted id can be
    // treated as a no-op abort
procedure abort(txn_id):
    aborted_before_start[txn_id] = true   // cheap, idempotent; see execute()
    txn = transactions_map[txn_id]
    if txn == null:
        // execute() hasn't registered this txn yet; it will observe the
        // tombstone right after it does (see execute()). Written before
        // this read so at least one of the two racing calls observes the
        // other, regardless of interleaving.
        return

    if txn.status == NULL and CAS(txn.status, NULL, abort_requested):
        // still executing on its own thread; let the executor clean itself
        // up once it notices, instead of touching its locks_acquired here
        // (single-cleanup-owner invariant)
        return

    if txn.status == abort_requested:
        // another thread already requested the abort; executor owns cleanup
        return

    abort_transaction(txn)


procedure mark_must_abort(txn):
    if CAS(txn.status, NULL, abort_requested):
        // still running: its own executor thread must discover this and
        // call abort_transaction on itself (single-cleanup-owner invariant)
        return
    if not CAS(txn.status, executed, must_abort):
        return
